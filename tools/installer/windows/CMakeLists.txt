cmake_minimum_required(VERSION 3.20)
project(mes_native_launcher LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This launcher uses the Windows service and process APIs")
endif()
add_executable(mes-homework WIN32 launcher.cpp)
set_target_properties(mes-homework PROPERTIES OUTPUT_NAME "MESHomework")
target_compile_features(mes-homework PRIVATE cxx_std_17)
target_compile_definitions(mes-homework PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(mes-homework PRIVATE advapi32 shell32 user32)
if(MSVC)
  target_compile_options(mes-homework PRIVATE /utf-8 /W4 /EHsc)
  set_property(TARGET mes-homework PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
enable_testing()
add_test(NAME native-launcher-arguments COMMAND mes-homework --self-test)
